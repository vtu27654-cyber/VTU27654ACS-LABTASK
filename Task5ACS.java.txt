
import java.util.*;

public class Task5ACS {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        int[] temp = new int[n];
        int[] result = new int[n];

        for (int i = 0; i < n; i++) {
            temp[i] = sc.nextInt();
            result[i] = -1;
        }

        Stack<Integer> stack = new Stack<>();

        for (int i = 0; i < n; i++) {
            while (!stack.isEmpty() && temp[i] > temp[stack.peek()]) {
                result[stack.pop()] = temp[i];
            }

            stack.push(i);
        }

        for (int i = 0; i < n; i++) {
            System.out.print(result[i]);

            if (i < n - 1)
                System.out.print(" ");
        }

        sc.close();
    }
}


**Sample Input**

```text
7
28 35 30 25 40 38 42
```

**Output**

```text
35 40 40 40 42 42 -1

**Output**


8
